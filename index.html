    // Terminal Logging Helper
    function logTerminal(text, type = 'info') {
        const log = document.createElement('div');
        if (type === 'success') {
            log.className = 'text-emerald-400';
            log.innerHTML = `[SUCCESS] ${text}`;

    // AI Chat System
    async function handleAISubmit() {
        const prompt = chatInput.value.trim();
        if (!prompt) return;

        if (state.currentUserRole === 'viewer') {
            alert('Permission Denied: Viewers cannot interact with AI code generation.');
            return;
        }

        // Append User Message
        appendChatMessage('user', prompt);
        chatInput.value = '';
        state.isThinking = true;
        updateAIThinkingState(true);

        // Build System Prompt with entire project context
        const currentProject = state.projects.find(p => p.id === state.currentProjectId);
        let contextPrompt = `You are Aether AI, an expert full-stack developer. You have access to the entire project workspace.

Here is the current project context:
`;
        
        currentProject.files.forEach(file => {
            contextPrompt += `\n--- FILE: ${file.name} ---
${file.content}
`;
        });

        contextPrompt += `\nWhen the user asks you to write, edit, or refactor code, you MUST output the updated or new files using the exact format:
[FILE: filename]
code here

You can output multiple files in a single response. Always maintain workspace memory and explain your changes briefly.`;

        try {
            const responseText = await window.AppHelpers.callAI(state.aiSettings, prompt, contextPrompt);
            
            // Append AI Response
            appendChatMessage('ai', responseText);
            
            // Parse response for file updates
            parseAIResponseAndApplyChanges(responseText);
            
            // Save conversation
            state.conversations.push({ role: 'user', content: prompt });
            state.conversations.push({ role: 'ai', content: responseText });
            window.AppHelpers.DB.set('conversations', state.conversations);

        } catch (error) {
            appendChatMessage('ai', `Error calling AI provider: ${error.message}. Falling back to local simulation.`);
            logTerminal(`AI Generation failed: ${error.message}`, 'error');
            window.AppHelpers.sound.playError();
        } finally {
            state.isThinking = false;
            updateAIThinkingState(false);
        }
    }

    function updateAIThinkingState(isThinking) {
        if (isThinking) {
            sendPromptBtn.innerHTML = ``;
            logTerminal('Aether AI is thinking & analyzing workspace context...', 'info');
        } else {
            sendPromptBtn.innerHTML = ``;
        }
        lucide.createIcons();
    }

    function appendChatMessage(role, text) {
        const msgEl = document.createElement('div');
        msgEl.className = `flex flex-col gap-1 p-3 rounded-lg text-xs ${role === 'user' ? 'bg-brand-600/10 border border-brand-500/20 self-end ml-8' : 'bg-zinc-900 border border-darkborder mr-8'}`;
        
        const header = role === 'user' ? 'You' : 'Aether AI';
        const headerColor = role === 'user' ? 'text-brand-400' : 'text-emerald-400';

        msgEl.innerHTML = `
            
    
                ${header}
                ${new Date().toLocaleTimeString()}
            
    
            
    ${escapeHtml(text)}
    
        `;
        chatMessages.appendChild(msgEl);
        chatMessages.scrollTop = chatMessages.scrollHeight;
    }

    function escapeHtml(text) {
        return text.replace(/&/g, '&').replace(//g, '>');
    }

    // Parse AI response and automatically update files
    function parseAIResponseAndApplyChanges(text) {
        const fileRegex = /\[FILE:\s*([a-zA-Z0-9_\-\.]+)\]\s*([\s\S]*?)(?=\[FILE:|$)/g;
        let match;
        let updatedAny = false;

        while ((match = fileRegex.exec(text)) !== null) {
            const filename = match[1].trim();
            let content = match[2].trim();

            // Clean up markdown code blocks if present
            if (content.startsWith('```')) {
                content = content.replace(/^```[a-zA-Z]*\n/, '').replace(/\n```$/, '');
            }

            updateFileContent(filename, content);
            logTerminal(`AI auto-updated file: ${filename}`, 'success');
            updatedAny = true;
        }

        if (updatedAny) {
            // Refresh editor if active file was updated
            const currentProject = state.projects.find(p => p.id === state.currentProjectId);
            const activeFileObj = currentProject.files.find(f => f.name === state.activeFile);
            if (state.editorInstance && activeFileObj) {
                state.editorInstance.setValue(activeFileObj.content);
            }
            runPreview();
        }
    }

                log.className = 'py-1 border-b border-darkborder/50';
                log.innerHTML = `${act.user} ${act.action}`;

    }

        // Sidebar Tabs
        document.querySelectorAll('.sidebar-tab-btn').forEach(btn => {
            btn.addEventListener('click', () => {
                switchSidebarTab(btn.getAttribute('data-tab'));
            });
        });

        // Run Preview
        runPreviewBtn.addEventListener('click', runPreview);
        refreshPreviewBtn.addEventListener('click', runPreview);
        toggleSplitBtn.addEventListener('click', togglePreviewPane);
        closePreviewBtn.addEventListener('click', togglePreviewPane);

        // Settings Modal
        settingsBtn.addEventListener('click', openSettings);
        closeSettingsBtn.addEventListener('click', () => settingsModal.classList.add('hidden'));
        cancelSettingsBtn.addEventListener('click', () => settingsModal.classList.add('hidden'));
        saveSettingsBtn.addEventListener('click', saveSettings);

        // Settings Modal Tabs
        document.querySelectorAll('.settings-tab-btn').forEach(btn => {
            btn.addEventListener('click', () => {
                document.querySelectorAll('.settings-tab-btn').forEach(b => b.classList.remove('active', 'border-brand-500', 'text-brand-400'));
                btn.classList.add('active', 'border-brand-500', 'text-brand-400');
                
                document.querySelectorAll('.settings-tab-content').forEach(c => c.classList.add('hidden'));
                document.getElementById(`settings-${btn.getAttribute('data-settings-tab')}-content`).classList.remove('hidden');
            });
        });

        // AI Chat Submit
        sendPromptBtn.addEventListener('click', handleAISubmit);
        chatInput.addEventListener('keypress', (e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                handleAISubmit();
            }
        });

        // Clear Chat
        clearChatBtn.addEventListener('click', () => {
            state.conversations = [];
            window.AppHelpers.DB.set('conversations', []);
            chatMessages.innerHTML = '';
            logTerminal('Chat history cleared.', 'info');
        });

        // Terminal Commands
        terminalInput.addEventListener('keypress', (e) => {
            if (e.key === 'Enter') {
                const cmd = terminalInput.value.trim();
                terminalInput.value = '';
                if (cmd) {
                    logTerminal(`$ ${cmd}`, 'info');
                    processTerminalCommand(cmd);
                }
            }
        });

        clearTerminalBtn.addEventListener('click', () => {
            terminalOutput.innerHTML = '';
        });

        // Auth Profile Switcher
        document.querySelectorAll('.auth-profile-btn').forEach(btn => {
            btn.addEventListener('click', () => {
                const role = btn.getAttribute('data-role');
                state.currentUserRole = role;
                userAvatar.textContent = role.charAt(0).toUpperCase();
                logTerminal(`Switched simulated user role to: ${role.toUpperCase()}`, 'info');
                
                // Update editor read-only state
                if (state.editorInstance) {
                    state.editorInstance.updateOptions({ readOnly: role === 'viewer' });
                }
                settingsModal.classList.add('hidden');
            });
        });
    }

    function processTerminalCommand(cmd) {
        const lower = cmd.toLowerCase();
        if (lower === 'help') {
            logTerminal('Available commands: npm run build, git status, deploy, clear, help', 'info');
        } else if (lower === 'npm run build') {
            logTerminal('Building production assets...', 'info');
            setTimeout(() => logTerminal('Build completed successfully!', 'success'), 1000);
        } else if (lower === 'git status') {
            logTerminal(`On branch ${state.githubSettings.branch || 'main'}\nYour branch is up to date.`, 'info');
        } else if (lower === 'deploy') {
            triggerDeployment();
        } else if (lower === 'clear') {
            terminalOutput.innerHTML = '';
        } else {
            logTerminal(`Command not found: ${cmd}. Type "help" for options.`, 'error');
        }
    }
